CREATE TABLE University (
    university_id INT PRIMARY KEY,
    university_name VARCHAR(100) NOT NULL,
    location VARCHAR(50) NOT NULL,
    state VARCHAR(50) NOT NULL,
    established_year INT
);

------------------------------------------------------

CREATE TABLE Department (
    department_id INT PRIMARY KEY,
    university_id INT NOT NULL,
    department_name VARCHAR(100) NOT NULL,
    office_location VARCHAR(100),
    hod_name VARCHAR(100),
    FOREIGN KEY (university_id) REFERENCES University(university_id)
);

------------------------------------------------------

CREATE TABLE Program (
    program_id INT PRIMARY KEY,
    department_id INT NOT NULL,
    program_name VARCHAR(100) NOT NULL,
    program_level VARCHAR(50),
    duration_years INT,
    FOREIGN KEY (department_id) REFERENCES Department(department_id)
);

------------------------------------------------------

CREATE TABLE Course (
    course_id INT PRIMARY KEY,
    program_id INT NOT NULL,
    course_name VARCHAR(100) NOT NULL,
    course_code VARCHAR(20) NOT NULL,
    credits INT,
    FOREIGN KEY (program_id) REFERENCES Program(program_id)
);

------------------------------------------------------

CREATE TABLE Syllabus (
    syllabus_id INT PRIMARY KEY,
    course_id INT NOT NULL,
    semester INT NOT NULL,
    academic_year VARCHAR(20),
    syllabus_details VARCHAR(100),
    FOREIGN KEY (course_id) REFERENCES Course(course_id)
);

------------------------------------------------------

CREATE TABLE Faculty (
    faculty_id INT PRIMARY KEY,
    department_id INT NOT NULL,
    faculty_name VARCHAR(100) NOT NULL,
    specialization VARCHAR(100),
    email VARCHAR(100),
    FOREIGN KEY (department_id) REFERENCES Department(department_id)
);

------------------------------------------------------

CREATE INDEX idx_university_location
ON University(location);

------------------------------------------------------

CREATE INDEX idx_department_university
ON Department(university_id);

------------------------------------------------------

CREATE INDEX idx_program_department
ON Program(department_id);

------------------------------------------------------

CREATE INDEX idx_course_program
ON Course(program_id);

------------------------------------------------------

CREATE INDEX idx_syllabus_course
ON Syllabus(course_id);

------------------------------------------------------

CREATE INDEX idx_faculty_specialization
ON Faculty(specialization);

SELECT *

SELECT
    d.department_id,
    d.department_name,
    d.office_location,
    d.hod_name
FROM Department d
JOIN University u
    ON d.university_id = u.university_id
WHERE u.university_name = 'Aligarh Muslim University';

------------------------------------------------------

SELECT
    university_name,
    location,
    state
FROM University
WHERE university_name = 'Jawaharlal Nehru University';

------------------------------------------------------

SELECT
    p.program_id,
    p.program_name,
    p.program_level,
    p.duration_years
FROM Program p
JOIN Department d
    ON p.department_id = d.department_id
JOIN University u
    ON d.university_id = u.university_id
WHERE u.university_name = 'University of Jammu';

------------------------------------------------------

SELECT DISTINCT
    u.university_id,
    u.university_name,
    u.location
FROM University u
JOIN Department d
    ON u.university_id = d.university_id
JOIN Program p
    ON d.department_id = p.department_id
WHERE p.program_name = 'MCA';

------------------------------------------------------

SELECT
    c.course_id,
    c.course_name,
    c.course_code,
    c.credits
FROM Course c
JOIN Program p
    ON c.program_id = p.program_id
JOIN Department d
    ON p.department_id = d.department_id
JOIN University u
    ON d.university_id = u.university_id
WHERE p.program_name = 'MCA'
AND u.university_name = 'Aligarh Muslim University';

------------------------------------------------------

SELECT
    f.faculty_name,
    f.specialization,
    f.email,
    u.university_name
FROM Faculty f
JOIN Department d
    ON f.department_id = d.department_id
JOIN University u
    ON d.university_id = u.university_id
WHERE f.specialization = 'Information Security';

------------------------------------------------------

SELECT
    u.university_name,
    c.course_name,
    s.semester,
    s.academic_year,
    s.syllabus_details
FROM Syllabus s
JOIN Course c
    ON s.course_id = c.course_id
JOIN Program p
    ON c.program_id = p.program_id
JOIN Department d
    ON p.department_id = d.department_id
JOIN University u
    ON d.university_id = u.university_id
WHERE c.course_name = 'Computer Architecture';
